# Database Setup Options
===== ===== ===== ===== =====
# This creates a new migration file in drizzle/ directory
> pnpm db:generate:dev

# HARD RESET (Destructive - wipes everything)
# Use when you want a completely fresh start
> pnpm reseed:hard:dev
# > ALLOW_ACCESS_TO_PRODUCTION_DB=true pnpm reseed:hard:prod

# SOFT RESET (Non-destructive - preserves data)
# Use when you want to apply migrations safely
> pnpm reseed:soft:dev
# > ALLOW_ACCESS_TO_PRODUCTION_DB=true pnpm reseed:soft:prod

# seed user data (for dev only)
> pnpm seed:user-data:medium-pareto:dev
# > ALLOW_ACCESS_TO_PRODUCTION_DB=true DISABLE_USER_SEEDING=false pnpm seed:user-data:medium-pareto:prod

> pnpm seed:test-user-interactions:clean:dev
# > ALLOW_ACCESS_TO_PRODUCTION_DB=true pnpm seed:test-user-interactions:clean:prod


# [Optional 1] Test clerk webhook locally
# for local clerk
> ngrok http --url=cuddly-unduly-zebra.ngrok-free.app 3000

# # [Optional 2] Test all triggers are working correctly
> pnpm db:test:all-triggers
# > pnpm db:test:all-triggers:prod:override

# # [Optional 3] View virtual db
> db:studio

# Future changes to schema
===== ===== ===== ===== =====
# 1. Edit schema files (src/lib/db/schema/*) or custom db files (src/lib/db/migrations/*)

# 2. Development - preserves data by default
> pnpm db:migrate:dev

# 3. Production - preserves data by default
> pnpm db:migrate:prod

# Alternative: Use soft reseed for complete setup with external data
> pnpm reseed:soft:dev

# 4. Test everything works
> pnpm db:test:integrity
> pnpm db:test:all-triggers
